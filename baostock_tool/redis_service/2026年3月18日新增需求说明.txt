配置需求：
行情源数据库配置
redis服务器配置
回测配置


回测业务流程：
1.按回测时间从数据库获取日线数据进行选股，将选股后数据推送到选股池，如未配置选股策略则将当日前10个证券推送到选股池；推送数据要符合消息结构要求
 "stocks": [
        {
            "symbol": "600036",
            "exchange": "SSE",
            "name": "招商银行",
            
            "basic_info": {
                "prev_close": 32.50,        // 昨收价
                "ma10": 31.80,               // 10日均线
                "ma5_high": 33               // 近5日最高价
                "volume_ratio": 1.2,         // 量比
                "turnover_rate": 0.5          // 换手率
            },
            
            "minute_volume_5d_01":            // 昨日每5分钟成交量（5天）
            [
                {"time": "09:35", "volume": 1234},
                {"time": "09:40", "volume": 2345},
                // ... 每5分钟一个点
            ],
            "minute_volume_5d_02":            
            [
                {"time": "09:35", "volume": 1234},
                {"time": "09:40", "volume": 2345},
                // ... 每5分钟一个点
            ],
            "technical_indicators": {
                "ma5": 31.20,
                "ma10": 31.80,
                "ma20": 32.10,
                "vol_ma5": 4567890,
                "vol_ma10": 5678901
            },
            
            "fundamental_data": {
                "pe": 8.5,
                "pb": 1.2,
                "market_cap": 820000  // 市值（百万）
            }            
        }，
        // ... 更多股票
    ]
}
2.当消费者消费了当日选股池后，查询对应日期tick数据表的对应证券代码数据，推送到行情通道；推送数据要符合消息结构要求：
消息结构
以json格式封装，定义如下：
{
    "type": "snapshot",
    "timestamp": 1703234567890,        // 毫秒时间戳
    "exchange": "SSE",                  // 交易所代码
    "symbol": "600036",                 // 股票代码
    "data": {
        "last_price": 32.56,             // 最新价
        "volume": 12345678,               // 成交量（手）
        "amount": 3987654321,             // 成交额
        "bid_price": [32.55, 32.54, ...], // 买一到买五价格
        "bid_volume": [100, 200, ...],    // 买一到买五量
        "ask_price": [32.57, 32.58, ...], // 卖一到卖五价格
        "ask_volume": [150, 250, ...],    // 卖一到卖五量
        “date”: "20260317"                // 行情日期
        "timestamp": "14:45:15"           // 行情时间
    }
}
