## 回测引擎概述

本项目支持两种回测引擎框架：

| 特性 | 时间遍历回测引擎 | Backtrader回测引擎 |
|------|------------------|-------------------|
| **入口文件** | `backtest_time_based_standard_mysqldb.py` | `backtrade_local_data_myst.py` |
| **基类** | `BaseStrategy` | `bt.Strategy` / `TPlus1Strategy` |
| **回测模式** | 按时间遍历，每日检查所有股票 | 单只股票独立回测 |
| **适用场景** | 多股票组合策略、资金管理 | 单股票策略验证、技术指标策略 |
| **数据来源** | MySQL数据库预加载 | MySQL数据库逐只加载 |
| **并发支持** | 单进程（数据预加载） | 多进程并发回测 |
| **T+1限制** | 内置支持 | 通过基类支持 |
| **动态止盈止损** | 基类内置方法 | 需自行实现 |

## 框架一：时间遍历回测引擎

### 架构说明

时间遍历回测引擎的工作流程：
```
1. 预加载所有股票数据到内存
2. 按交易日顺序遍历
3. 每个交易日：
   - 检查所有持仓的卖出信号
   - 检查所有股票的买入信号
   - 按信号强度排序买入
4. 记录每日资产变化
5. 计算统计指标
```

### BaseStrategy 基类详解

所有时间遍历策略必须继承 `BaseStrategy` 类，该类定义在 `base_strategy.py` 中。

#### 类属性

```python
class BaseStrategy(ABC):
    # 策略名称（子类必须设置）
    STRATEGY_NAME = "BaseStrategy"
    
    # 策略参数默认值（子类可覆盖）
    DEFAULT_PARAMS = {}
    
    # 动态止盈止损默认参数（子类可覆盖）
    DEFAULT_STOP_PARAMS = {
        'take_profit': 0.12,           # 基础止盈点：12%
        'stop_loss': -0.06,            # 基础止损点：-6%
        'trailing_stop_trigger': 0.06, # 触发移动止损的盈利阈值：6%
        'trailing_stop_pct': 0.03,     # 移动止损回撤比例：3%
        'trailing_stop_min_profit': 0.02,  # 移动止损最低保底盈利：2%
    }
```

def check_buy_signal(
    self, 
    stock_code: str,
    market: str,
    stock_data: pd.DataFrame,
    current_date: str
) -> Tuple[bool, float, Dict[str, Any]]:
    """
    买入信号检测逻辑
    """
    try:
        # 1. 数据有效性检查
        min_length = self.get_min_data_length()
        if stock_data is None or len(stock_data) < min_length:
            return False, 0, {'reason': f'数据不足{min_length}天'}
        
        # 2. 获取当日数据
        current_date_dt = pd.to_datetime(current_date)
        if current_date_dt not in stock_data.index:
            return False, 0, {'reason': '当日无数据'}
        
        today_data = stock_data.loc[current_date_dt]
        
        # 3. 计算指标（示例：简单移动平均）
        close_prices = stock_data['close']
        ma_value = close_prices.rolling(window=self.params['param1']).mean().iloc[-1]
        
        # 4. 判断买入条件
        if today_data['close'] > ma_value * (1 + self.params['param2']):
            # 满足条件，返回买入信号
            signal_strength = (today_data['close'] - ma_value) / ma_value
            signal_info = {
                'close_price': float(today_data['close']),  # 必须！
                'ma_value': float(ma_value),
                # ... 更多信息
            }
            return True, signal_strength, signal_info
        
        # 不满足条件
        return False, 0, {'reason': '不满足买入条件'}
        
    except Exception as e:
        return False, 0, {'error': str(e)}

### 完整示例

以下是一个完整的价值投资策略示例（简化版）：

```python
"""
价值投资策略

买入条件：
1. PE < 30 且 PB < 12
2. 当日最低价比210日均线低15%以上

卖出条件：
1. 当日最高价比210日均线高25%以上
2. 持仓达到120天
3. 动态止盈止损触发
"""

from typing import Dict, Any, Tuple, Optional
import pandas as pd
import numpy as np
from .base_strategy import BaseStrategy, Position


class ValueStrategyTimeBased(BaseStrategy):
    """价值投资策略（按时间遍历版本）"""
    
    STRATEGY_NAME = "ValueStrategy"
    
    DEFAULT_PARAMS = {
        'ma_period': 210,           # 均线周期
        'max_pe_ttm': 30.0,         # 最大市盈率
        'max_pb_mrq': 12.0,         # 最大市净率
        'buy_threshold': -0.15,     # 买入阈值：最低价比MA低15%
        'sell_threshold_up': 0.25,  # 卖出阈值：最高价比MA高25%
        'max_hold_days': 120,       # 最大持仓天数
        # 覆盖止盈止损参数
        'take_profit': 0.26,
        'stop_loss': -0.06,
        'trailing_stop_trigger': 0.10,
        'trailing_stop_pct': 0.05,
    }
    
    def _validate_params(self):
        """验证参数有效性"""
        assert 0 < self.params['max_pe_ttm'] < 1000
        assert 0 < self.params['max_pb_mrq'] < 1000
        assert -1 < self.params['buy_threshold'] < 0
        assert 0 < self.params['sell_threshold_up'] < 1
    
    def get_min_data_length(self) -> int:
        """需要至少MA周期的数据"""
        return self.params['ma_period']
    
    def get_lookback_days(self) -> int:
        """需要预加载 MA 周期的数据用于计算均线"""
        return self.params['ma_period']
    
    def check_buy_signal(
        self, 
        stock_code: str,
        market: str,
        stock_data: pd.DataFrame,
        current_date: str
    ) -> Tuple[bool, float, Dict[str, Any]]:
        """检查买入信号"""
        try:
            ma_period = self.params['ma_period']
            
            # 数据检查
            if stock_data is None or len(stock_data) < ma_period:
                return False, 0, {'reason': f'数据不足{ma_period}天'}
            
            # 获取当日数据
            current_date_dt = pd.to_datetime(current_date)
            if current_date_dt not in stock_data.index:
                return False, 0, {'reason': '当日无数据'}
            
            today_data = stock_data.loc[current_date_dt]
            
            # 检查估值条件
            pe_ttm = today_data.get('peTTM', None)
            pb_mrq = today_data.get('pbMRQ', None)
            
            if pe_ttm is None or pe_ttm <= 0 or pe_ttm >= self.params['max_pe_ttm']:
                return False, 0, {'reason': f'PE不满足: {pe_ttm}'}
            
            if pb_mrq is None or pb_mrq <= 0 or pb_mrq >= self.params['max_pb_mrq']:
                return False, 0, {'reason': f'PB不满足: {pb_mrq}'}
            
            # 计算均线
            close_prices = stock_data['close']
            ma_value = close_prices.rolling(window=ma_period).mean().iloc[-1]
            
            if pd.isna(ma_value) or ma_value <= 0:
                return False, 0, {'reason': '均线无效'}
            
            # 检查买入条件
            low_price = today_data['low']
            deviation = (low_price - ma_value) / ma_value
            
            if deviation <= self.params['buy_threshold']:
                signal_strength = abs(deviation)
                signal_info = {
                    'close_price': float(today_data['close']),
                    'ma_value': float(ma_value),
                    'deviation': float(deviation),
                    'pe_ttm': float(pe_ttm),
                    'pb_mrq': float(pb_mrq),
                }
                return True, signal_strength, signal_info
            
            return False, 0, {'reason': f'偏离度不满足: {deviation:.2%}'}
            
        except Exception as e:
            return False, 0, {'error': str(e)}
    
    def check_sell_signal(
        self,
        position: Position,
        stock_data: pd.DataFrame,
        current_date: str
    ) -> Tuple[bool, Optional[str], Optional[float]]:
        """检查卖出信号"""
        try:
            current_date_dt = pd.to_datetime(current_date)
            if current_date_dt not in stock_data.index:
                return False, None, None
            
            today_data = stock_data.loc[current_date_dt]
            current_price = today_data['close']
            
            # 1. 动态止盈止损
            should_stop, stop_reason = self.check_dynamic_stop_loss(position, current_price)
            if should_stop:
                return True, stop_reason, current_price
            
            # 2. 最高价触及MA+25%
            ma_period = self.params['ma_period']
            close_prices = stock_data['close']
            if len(close_prices) >= ma_period:
                ma_value = close_prices.rolling(window=ma_period).mean().iloc[-1]
                if not pd.isna(ma_value) and ma_value > 0:
                    high_price = today_data['high']
                    if high_price >= ma_value * (1 + self.params['sell_threshold_up']):
                        return True, f'最高价触及MA+{self.params["sell_threshold_up"]*100:.0f}%', current_price
            
            # 3. 持仓天数达到最大
            if position.hold_days >= self.params['max_hold_days']:
                return True, f'持仓达到{self.params["max_hold_days"]}天', today_data['open']
            
            return False, None, None
            
        except Exception as e:
            return False, None, None
```

## 框架二：Backtrader回测引擎

### 架构说明

Backtrader回测引擎的工作流程：
```
1. 逐只股票加载数据
2. 创建 Cerebro 引擎
3. 添加策略类和数据源
4. 运行回测
5. 收集分析结果
6. 保存触发点位
```

Backtrader框架的核心概念：
- **Cerebro**: 回测引擎核心，管理策略、数据、分析器
- **Strategy**: 策略类，定义交易逻辑
- **Data Feed**: 数据源，提供OHLCV数据
- **Analyzer**: 分析器，计算各种指标
- **Broker**: 经纪人，管理资金和订单

    self.order = None

    self.total_commission = 0.0
    self.buy_commission = 0.0
    self.sell_commission = 0.0

#### 第四步：实现 notify_order 方法

```python
def notify_order(self, order):
    """订单状态通知"""
    # 忽略提交和接受状态的订单
    if order.status in [order.Submitted, order.Accepted]:
        return
    
    # 订单完成
    if order.status in [order.Completed]:
        if order.isbuy():
            # 买入完成
            self.buyprice = order.executed.price
            self.buycomm = order.executed.comm
            self.buy_date = self.datas[0].datetime.date()
            
            # 统计手续费
            self.buy_commission += order.executed.comm
            self.total_commission += order.executed.comm
            
            # 记录买入触发点位
            buy_info = {
                'date': str(self.buy_date),
                'trigger_type': 'buy',
                'price': float(order.executed.price),
                'volume': float(order.executed.size),
                'commission': float(order.executed.comm)
            }
            self.current_buy_info = buy_info
            
            if self.p.printlog:
                logger.debug(f'买入执行 - 价格: {order.executed.price:.2f}, '
                          f'数量: {order.executed.size}, '
                          f'手续费: {order.executed.comm:.2f}')
        else:
            # 卖出完成
            self.sell_commission += order.executed.comm
            self.total_commission += order.executed.comm
            
            # 记录卖出触发点位
            if self.current_buy_info:
                sell_date = self.datas[0].datetime.date()
                sell_info = {
                    'date': str(sell_date),
                    'trigger_type': 'sell',
                    'price': float(order.executed.price),
                    'volume': float(order.executed.size),
                    'commission': float(order.executed.comm),
                    'profit': float(order.executed.pnl),
                    'profit_rate': round((order.executed.pnl / self.buyprice) * 100, 2) if self.buyprice else 0
                }
                self.trigger_points.append(self.current_buy_info)
                self.trigger_points.append(sell_info)
                self.current_buy_info = None
            
            if self.p.printlog:
                logger.debug(f'卖出执行 - 价格: {order.executed.price:.2f}, '
                          f'数量: {order.executed.size}, '
                          f'手续费: {order.executed.comm:.2f}, '
                          f'利润: {order.executed.pnl:.2f}')
            self.buy_date = None
    
    # 订单取消/拒绝
    elif order.status in [order.Canceled, order.Margin, order.Rejected]:
        if self.p.printlog:
            logger.warning(f'订单取消/拒绝 - 状态: {order.getstatusname()}')
    
    self.order = None
```

#### 第五步：实现 notify_trade 方法

```python
def notify_trade(self, trade):
    """交易完成通知"""
    if not trade.isclosed:
        return
    
    # 统计盈利和亏损交易
    if trade.pnlcomm > 0:
        self.profit_trade_count += 1
    elif trade.pnlcomm < 0:
        self.loss_trade_count += 1
    
    if self.p.printlog:
        logger.debug(f'交易利润 - 毛利润: {trade.pnl:.2f}, 净利润: {trade.pnlcomm:.2f}')
```

#### 第六步：实现 next 方法（核心逻辑）

```python
def next(self):
    """主逻辑"""
    # 如果有待处理订单，跳过
    if self.order:
        return
    
    # 如果有持仓，检查卖出条件
    if self.position:
        current_price = self.dataclose[0]
        
        # 计算盈亏
        if self.buyprice > 0:
            profit_rate = ((current_price - self.buyprice) / self.buyprice) * 100
            
            # 卖出条件
            if profit_rate >= 10 or profit_rate <= -5:
                self.order = self.sell(size=self.position.size)
        return
    
    # 无持仓，检查买入条件
    # 示例：收盘价上穿均线
    if self.dataclose[-1] < self.sma[-1] and self.dataclose[0] > self.sma[0]:
        # 计算买入数量
        cash = self.broker.getvalue()
        price = self.dataclose[0]
        if price > 0:
            buy_size = int((cash * 0.90) / price)  # 90%仓位
            if buy_size > 0:
                self.order = self.buy(size=buy_size)
```

### 完整示例

以下是一个完整的底分型策略示例：

```python
import backtrader as bt
from utils.logger_utils import setup_logger
import config

log_config = config.get_log_config()
logger = setup_logger(
    logger_name=__name__,
    log_level=log_config["log_level"],
    log_dir=log_config["log_dir"]
)

class BottomFractalStrategy(bt.Strategy):
    """
    底分型策略：
    1. 检查K线是否形成底分型（中间K线最低）
    2. 出现底分型后，下一交易日以开盘价买入
    3. 设置止盈、止损和持仓天数限制
    
    底分型定义：
    - 连续三根K线：K1(前天), K2(昨天), K3(今天)
    - 条件：K2的最低价 < K1的最低价 且 K2的最低价 < K3的最低价
    """
    
    STRATEGY_NAME = "底分型策略"
    
    params = (
        ('stock_code', ''),
        ('profit_threshold', 10.0),      # 止盈阈值（%）
        ('drawdown_threshold', 3.0),     # 止损阈值（%）
        ('hold_days_threshold', 99999),  # 最大持仓天数
        ('printlog', True),
    )
    
    def __init__(self):
        # 数据访问
        self.dataclose = self.datas[0].close
        self.dataopen = self.datas[0].open
        self.datalow = self.datas[0].low
        self.datahigh = self.datas[0].high
        
        # 订单和状态
        self.order = None
        self.buyprice = None
        self.buycomm = None
        self.buy_date = None
        self.hold_days = 0
        
        # 统计变量
        self.total_commission = 0.0
        self.buy_commission = 0.0
        self.sell_commission = 0.0
        self.profit_trade_count = 0
        self.loss_trade_count = 0
        
        # 触发点位
        self.trigger_points = []
        self.current_buy_info = None
        self.last_trigger_dfx_info = None
    
    def is_bottom_fractal(self):
        """判断是否形成底分型"""
        if len(self.datalow) < 3:
            return False, None
        
        low_t3 = self.datalow[-3]  # 前天最低价
        low_t2 = self.datalow[-2]  # 昨天最低价
        low_t1 = self.datalow[-1]  # 今天最低价
        
        # 底分型：昨天最低价 < 前天最低价 且 昨天最低价 < 今天最低价
        if low_t2 < low_t3 and low_t2 < low_t1:
            date_t3 = self.datas[0].datetime.date(-3)
            date_t2 = self.datas[0].datetime.date(-2)
            date_t1 = self.datas[0].datetime.date(-1)
            
            trigger_info = {
                'date_t3': str(date_t3),
                'low_t3': float(low_t3),
                'date_t2': str(date_t2),
                'low_t2': float(low_t2),
                'date_t1': str(date_t1),
                'low_t1': float(low_t1)
            }
            return True, trigger_info
        
        return False, None
    
    def notify_order(self, order):
        """订单状态通知"""
        if order.status in [order.Submitted, order.Accepted]:
            return
        
        if order.status in [order.Completed]:
            if order.isbuy():
                self.buyprice = order.executed.price
                self.buycomm = order.executed.comm
                self.buy_date = self.datas[0].datetime.date()
                self.hold_days = 1
                
                self.buy_commission += order.executed.comm
                self.total_commission += order.executed.comm
                
                buy_info = {
                    'date': str(self.buy_date),
                    'trigger_type': 'buy',
                    'price': float(order.executed.price),
                    'volume': float(order.executed.size),
                    'commission': float(order.executed.comm),
                    'volume_rank': '底分型',
                }
                self.current_buy_info = buy_info
                self.last_trigger_dfx_info = None
            else:
                self.sell_commission += order.executed.comm
                self.total_commission += order.executed.comm
                
                if self.current_buy_info:
                    sell_date = self.datas[0].datetime.date()
                    sell_info = {
                        'date': str(sell_date),
                        'trigger_type': 'sell',
                        'price': float(order.executed.price),
                        'volume': float(order.executed.size),
                        'commission': float(order.executed.comm),
                        'profit': float(order.executed.pnl),
                        'profit_rate': round((order.executed.pnl / self.buyprice) * 100, 2) if self.buyprice else 0,
                        'hold_days': self.hold_days
                    }
                    self.trigger_points.append(self.current_buy_info)
                    self.trigger_points.append(sell_info)
                    self.current_buy_info = None
                
                self.buy_date = None
                self.hold_days = 0
        
        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            if self.p.printlog:
                logger.warning(f'订单取消/拒绝 - 状态: {order.getstatusname()}')
        
        self.order = None
    
    def notify_trade(self, trade):
        """交易完成通知"""
        if not trade.isclosed:
            return
        
        if trade.pnlcomm > 0:
            self.profit_trade_count += 1
        elif trade.pnlcomm < 0:
            self.loss_trade_count += 1
    
    def stop(self):
        """回测结束时的处理"""
        if self.current_buy_info:
            self.trigger_points.append(self.current_buy_info)
            if self.p.printlog:
                logger.warning(f'回测结束，存在未平仓持仓')
    
    def next(self):
        """主逻辑"""
        if self.order:
            return
        
        # 有持仓，检查卖出
        if self.position:
            self.hold_days += 1
            current_price = self.dataclose[0]
            
            if self.buyprice > 0:
                profit_rate = ((current_price - self.buyprice) / self.buyprice) * 100
                
                # 卖出条件
                if (profit_rate >= self.p.profit_threshold or 
                    profit_rate <= -self.p.drawdown_threshold or 
                    self.hold_days >= self.p.hold_days_threshold):
                    
                    if self.p.printlog:
                        logger.info(f'触发卖出 - 价格: {current_price:.2f}, '
                                  f'浮盈: {profit_rate:.2f}%, '
                                  f'持仓: {self.hold_days}天')
                    
                    self.order = self.sell(size=self.position.size)
            return
        
        # 无持仓，检查买入
        is_dfx, dfx_info = self.is_bottom_fractal()
        
        if is_dfx:
            self.last_trigger_dfx_info = dfx_info
            
            if self.p.printlog:
                logger.info(f'触发买入 - 底分型形成')
            
            cash = self.broker.getvalue()
            price = self.dataclose[-1]
            if price > 0:
                buy_size = int((cash * 0.90) / price)
                if buy_size > 0:
                    self.order = self.buy(size=buy_size)
```

### Q1: 如何选择回测引擎？

**时间遍历引擎**适合：
- 多股票组合策略
- 需要资金管理和仓位控制
- 需要动态止盈止损
- 需要考虑股票间的资金分配

**Backtrader引擎**适合：
- 单股票策略验证
- 技术指标策略
- 需要使用Backtrader的丰富分析器
- 需要多进程并发回测
